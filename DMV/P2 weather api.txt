# Import required libraries
import requests
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import seaborn as sns
from getpass import getpass
import matplotlib.dates as mdates


# Define a function to fetch and analyze weather data
def fetch_and_analyze_weather_data():

    # Step 1: Enter API key securely
    # getpass hides the API key while entering it
    API_KEY = getpass("Enter OpenWeatherMap API Key: ")

    # Specify city and country
    city = "Pune"
    country = "IN"

    # OpenWeatherMap 5-day weather forecast API URL
    url = "https://api.openweathermap.org/data/2.5/forecast"

    # Parameters required to make the API request
    params = {
        "q": f"{city},{country}",  # City and country
        "appid": API_KEY,          # API authentication key
        "units": "metric"          # Temperature in Celsius
    }

    try:
        # Step 2: Fetch data from API
        # Send GET request with parameters
        response = requests.get(url, params=params, timeout=20)

        # Check if API key is invalid or not activated
        if response.status_code == 401:
            print("Invalid API key or API key not activated.")
            print(response.text)
            return

        # Raise an exception if HTTP error occurs
        response.raise_for_status()

        # Convert JSON API response into Python dictionary
        data = response.json()

        # Display API response information
        print("\nAPI Request Successful!")
        print("City:", data["city"]["name"])
        print("Country:", data["city"]["country"])
        print("Latitude:", data["city"]["coord"]["lat"])
        print("Longitude:", data["city"]["coord"]["lon"])


        # Step 3: Extract weather information

        # Create an empty list to store weather records
        records = []

        # Iterate through each weather forecast record
        for item in data["list"]:

            # Extract required weather details
            records.append({
                "Date": item["dt_txt"],
                "Temperature": item["main"]["temp"],
                "Humidity": item["main"]["humidity"],
                "Wind_Speed": item["wind"]["speed"],

                # Extract rainfall for 3 hours
                # If rainfall data is unavailable, use 0
                "Rainfall": item.get("rain", {}).get("3h", 0),

                # Extract weather description
                "Weather": item["weather"][0]["description"]
            })


        # Step 4: Convert into DataFrame

        # Convert the list of records into a Pandas DataFrame
        weather = pd.DataFrame(records)

        # Convert Date column into datetime format
        weather["Date"] = pd.to_datetime(weather["Date"])

        # Define columns containing numeric values
        numeric_columns = [
            "Temperature",
            "Humidity",
            "Wind_Speed",
            "Rainfall"
        ]


        # Step 5: Data cleaning

        # Convert numeric columns into numeric data types
        for col in numeric_columns:
            weather[col] = pd.to_numeric(
                weather[col], errors="coerce"
            )

            # Replace missing rainfall values with 0
            if col == "Rainfall":
                weather[col] = weather[col].fillna(0)

            # Replace missing values in other numeric columns
            # with their respective median values
            else:
                weather[col] = weather[col].fillna(
                    weather[col].median()
                )

        # Remove records having missing dates
        weather = weather.dropna(subset=["Date"])

        # Display first 10 weather records
        print("\nRetrieved Weather Data:\n")
        print(weather.head(10).to_string(index=False))

        # Display missing value count in each column
        print("\nMissing Values:\n")
        print(weather.isnull().sum())


        # Step 6: Statistical analysis

        # Display descriptive statistics of numeric columns
        print("\nWeather Statistics:\n")
        print(weather[numeric_columns].describe())

        # Calculate average temperature
        print("\nAverage Temperature:",
              round(weather["Temperature"].mean(), 2), "°C")

        # Find maximum temperature
        print("Maximum Temperature:",
              weather["Temperature"].max(), "°C")

        # Find minimum temperature
        print("Minimum Temperature:",
              weather["Temperature"].min(), "°C")

        # Calculate average humidity
        print("Average Humidity:",
              round(weather["Humidity"].mean(), 2), "%")

        # Calculate total rainfall
        print("Total Rainfall:",
              round(weather["Rainfall"].sum(), 2), "mm")


        # Step 7: Daily weather aggregation

        # Extract only the date from datetime values
        weather["Date_Only"] = weather["Date"].dt.date

        # Group weather records by date
        # Calculate daily average temperature, humidity and wind speed
        # Calculate total rainfall for each day
        daily_data = weather.groupby("Date_Only").agg({
            "Temperature": "mean",
            "Humidity": "mean",
            "Wind_Speed": "mean",
            "Rainfall": "sum"
        }).round(2)

        # Display daily weather summary
        print("\nDaily Weather Summary:\n")
        print(daily_data)


        # Step 8: Save data into CSV files

        # Save complete weather data into CSV file
        weather.to_csv("openweather_data.csv", index=False)

        # Save daily aggregated weather data into CSV file
        daily_data.to_csv("daily_weather_summary.csv")

        print("\nWeather data saved successfully!")


        # Step 9: Visualize weather data

        # Create 4 subplots arranged in 2 rows and 2 columns
        fig, axes = plt.subplots(
            2, 2, figsize=(15, 10),
            constrained_layout=True
        )

        # Set common date formatting for all graphs
        date_locator = mdates.AutoDateLocator()
        date_formatter = mdates.ConciseDateFormatter(date_locator)


        # Temperature graph

        # Plot temperature changes over time
        # marker="o" displays circular markers
        axes[0, 0].plot(
            weather["Date"],
            weather["Temperature"],
            marker="o"
        )

        axes[0, 0].set_title("Temperature Trend")
        axes[0, 0].set_xlabel("Date")
        axes[0, 0].set_ylabel("Temperature (°C)")

        # Format date labels on X-axis
        axes[0, 0].xaxis.set_major_locator(date_locator)
        axes[0, 0].xaxis.set_major_formatter(date_formatter)

        # Display grid lines
        axes[0, 0].grid(True, alpha=0.3)


        # Humidity graph

        # Create a bar chart showing humidity values
        axes[0, 1].bar(
            weather["Date"],
            weather["Humidity"],
            width=0.08
        )

        axes[0, 1].set_title("Humidity")
        axes[0, 1].set_xlabel("Date")
        axes[0, 1].set_ylabel("Humidity (%)")

        # Format date labels
        axes[0, 1].xaxis.set_major_locator(date_locator)
        axes[0, 1].xaxis.set_major_formatter(date_formatter)


        # Wind speed graph

        # Plot wind speed changes over time
        axes[1, 0].plot(
            weather["Date"],
            weather["Wind_Speed"],
            marker="o"
        )

        axes[1, 0].set_title("Wind Speed")
        axes[1, 0].set_xlabel("Date")
        axes[1, 0].set_ylabel("Wind Speed (m/s)")

        # Format date labels
        axes[1, 0].xaxis.set_major_locator(date_locator)
        axes[1, 0].xaxis.set_major_formatter(date_formatter)

        # Display grid lines
        axes[1, 0].grid(True, alpha=0.3)


        # Rainfall graph

        # Create a bar chart showing rainfall values
        axes[1, 1].bar(
            weather["Date"],
            weather["Rainfall"],
            width=0.08
        )

        axes[1, 1].set_title("Rainfall")
        axes[1, 1].set_xlabel("Date")
        axes[1, 1].set_ylabel("Rainfall (mm)")

        # Format date labels
        axes[1, 1].xaxis.set_major_locator(date_locator)
        axes[1, 1].xaxis.set_major_formatter(date_formatter)


        # Handle zero rainfall

        # Check whether all rainfall values are zero
        if weather["Rainfall"].max() == 0:

            # Set Y-axis limits to avoid an empty-looking graph
            axes[1, 1].set_ylim(0, 1)

            # Display a message in the rainfall graph
            axes[1, 1].text(
                0.5, 0.5,
                "No rainfall predicted",
                transform=axes[1, 1].transAxes,
                ha="center",
                va="center"
            )

        # Add the main title to all four graphs
        plt.suptitle("Weather Data Analysis - Pune", fontsize=16)

        # Display the graphs
        plt.show()


        # Step 10: Correlation analysis

        # Calculate correlation between numeric weather columns
        correlation = weather[numeric_columns].corr()

        # Remove constant columns from correlation analysis
        # A column must contain more than one unique value
        variable_columns = [
            col for col in numeric_columns
            if weather[col].nunique() > 1
        ]

        # Display the correlation matrix
        print("\nCorrelation Matrix:\n")
        print(correlation.loc[
            variable_columns, variable_columns
        ].round(4))


        # Step 11: Correlation heatmap

        # Create a figure for the heatmap
        plt.figure(figsize=(8, 5))

        # Create a heatmap to visualize correlations
        sns.heatmap(
            correlation.loc[variable_columns, variable_columns],
            annot=True,          # Display correlation values
            cmap="coolwarm",     # Set color scheme
            fmt=".2f",           # Display values up to 2 decimals
            linewidths=0.5       # Add space between cells
        )

        # Add title to heatmap
        plt.title("Weather Correlation Heatmap")

        # Adjust layout
        plt.tight_layout()

        # Display heatmap
        plt.show()


        # Display excluded columns, if any

        # Identify columns removed from correlation analysis
        excluded_columns = [
            col for col in numeric_columns
            if col not in variable_columns
        ]

        # Display excluded columns and reason
        if excluded_columns:
            print("\nExcluded from correlation analysis:")
            print(excluded_columns)
            print("Reason: These columns contain constant values.")

        print("\nWeather Data Analysis Completed Successfully!")


    # Handle HTTP errors such as invalid API requests
    except requests.exceptions.HTTPError as e:
        print("HTTP Error:", e)
        print("Response:", response.text)

    # Handle network errors and connection problems
    except requests.exceptions.RequestException as e:
        print("Network Error:", e)

    # Handle errors related to missing keys or invalid data
    except (KeyError, ValueError, TypeError) as e:
        print("Data Processing Error:", e)


# Execute the function
fetch_and_analyze_weather_data()
