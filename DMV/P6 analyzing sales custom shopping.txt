# PRACTICAL 6: DATA AGGREGATION

# Import required libraries
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt


# --------------------------------------------------
# 1. LOAD DATASET
# --------------------------------------------------

# Read the customer shopping dataset
data = pd.read_csv("customer_shopping_data.csv")

# Display successful loading message
print("Dataset loaded successfully!")

# Display number of rows and columns
print("\nDataset Shape:", data.shape)

# Display first 5 records
print("\nFirst 5 Records:")
print(data.head())

# Display dataset information like columns and data types
print("\nDataset Information:")
data.info()

# Display statistical summary of numerical columns
print("\nStatistical Summary:")
print(data.describe())

# Display all column names
print("\nColumn Names:")
print(data.columns.tolist())


# --------------------------------------------------
# 2. CHECK MISSING VALUES AND DUPLICATES
# --------------------------------------------------

# Count missing values in each column
print("\nMissing Values:")
print(data.isnull().sum())

# Count duplicate records before removal
print("\nDuplicate Records Before Removal:")
print(data.duplicated().sum())

# Remove duplicate records permanently
data.drop_duplicates(inplace=True)

# Check duplicate records after removal
print("\nDuplicate Records After Removal:")
print(data.duplicated().sum())


# --------------------------------------------------
# 3. DATA CLEANING AND PREPARATION
# --------------------------------------------------

# Convert quantity column into numeric format
data["quantity"] = pd.to_numeric(data["quantity"], errors="coerce")

# Convert price column into numeric format
data["price"] = pd.to_numeric(data["price"], errors="coerce")

# Convert age column into numeric format
data["age"] = pd.to_numeric(data["age"], errors="coerce")

# Remove records having missing values in essential columns
data.dropna(
    subset=["quantity", "price", "category",
            "shopping_mall", "payment_method", "invoice_date"],
    inplace=True
)

# Fill missing age values with the median age
data["age"] = data["age"].fillna(data["age"].median())

# Convert invoice_date column into datetime format
# dayfirst=True means date is in day/month/year format
data["invoice_date"] = pd.to_datetime(
    data["invoice_date"],
    dayfirst=True,
    errors="coerce"
)

# Remove records having invalid dates
data.dropna(subset=["invoice_date"], inplace=True)

# Display total missing values after cleaning
print("\nMissing Values After Treatment:")
print(data.isnull().sum().sum())

# Display cleaned dataset shape
print("\nCleaned Dataset Shape:", data.shape)


# --------------------------------------------------
# 4. CALCULATE SALES AMOUNT
# --------------------------------------------------

# Calculate sales amount for each transaction
# Sales Amount = Quantity * Price
data["sales_amount"] = data["quantity"] * data["price"]

print("\nSales Amount Calculated Successfully!")

# Display updated dataset
print("\nUpdated Dataset:")
print(data.head())


# --------------------------------------------------
# 5. EXTRACT MONTH AND YEAR
# --------------------------------------------------

# Extract year and month from invoice_date
# to_period("M") gives year-month format like 2022-08
data["month"] = data["invoice_date"].dt.to_period("M").astype(str)

# Display invoice date and extracted month
print("\nUpdated Dataset with Month:")
print(data[["invoice_date", "month"]].head())


# --------------------------------------------------
# 6. AGGREGATION BY PRODUCT CATEGORY
# --------------------------------------------------

# Group data by product category
# Calculate total sales, total quantity and number of transactions
category_sales = data.groupby("category").agg(
    total_sales=("sales_amount", "sum"),
    total_quantity=("quantity", "sum"),
    total_transactions=("invoice_no", "count")
).sort_values(by="total_sales", ascending=False)

# Display category-wise sales summary
print("\nTotal Sales by Product Category:")
print(category_sales)


# --------------------------------------------------
# 7. AGGREGATION BY SHOPPING MALL
# --------------------------------------------------

# Group data by shopping mall
# Calculate total sales and total quantity for each mall
mall_sales = data.groupby("shopping_mall").agg(
    total_sales=("sales_amount", "sum"),
    total_quantity=("quantity", "sum")
).sort_values(by="total_sales", ascending=False)

# Display mall-wise sales summary
print("\nTotal Sales by Shopping Mall:")
print(mall_sales)


# --------------------------------------------------
# 8. AGGREGATION BY PAYMENT METHOD
# --------------------------------------------------

# Group data by payment method
# Calculate total sales for each payment method
payment_sales = data.groupby("payment_method")["sales_amount"].sum()

# Display payment method-wise sales
print("\nSales by Payment Method:")
print(payment_sales)


# --------------------------------------------------
# 9. MONTHLY SALES AGGREGATION
# --------------------------------------------------

# Group data by month and calculate total monthly sales
monthly_sales = data.groupby("month")["sales_amount"].sum()

# Display monthly sales
print("\nMonthly Sales:")
print(monthly_sales)


# --------------------------------------------------
# 10. PIVOT TABLE: SHOPPING MALL VS CATEGORY
# --------------------------------------------------

# Create a pivot table
# Rows represent shopping malls
# Columns represent product categories
# Values represent total sales
pivot_table = pd.pivot_table(
    data,
    values="sales_amount",
    index="shopping_mall",
    columns="category",
    aggfunc="sum",
    fill_value=0
)

# Display pivot table
print("\nPivot Table (Shopping Mall vs Category):")
print(pivot_table)


# --------------------------------------------------
# 11. CUSTOMER GENDER-WISE SALES
# --------------------------------------------------

# Group data by gender and calculate total sales
gender_sales = data.groupby("gender")["sales_amount"].sum()

# Display gender-wise sales
print("\nGender-wise Sales:")
print(gender_sales)


# --------------------------------------------------
# 12. DISPLAY TOP CATEGORY AND TOP MALL
# --------------------------------------------------

# Find the category having the highest total sales
top_category = category_sales["total_sales"].idxmax()

# Find the shopping mall having the highest total sales
top_mall = mall_sales["total_sales"].idxmax()

# Calculate total revenue from all transactions
total_revenue = data["sales_amount"].sum()

# Display top-selling category, mall and total revenue
print("\nTop Selling Category:", top_category)
print("Top Performing Shopping Mall:", top_mall)
print("Total Revenue: {:.2f}".format(total_revenue))


# --------------------------------------------------
# 13. VISUALIZATION
# --------------------------------------------------

# A. Sales by Product Category

# Set graph size
plt.figure(figsize=(10, 5))

# Create bar chart for total sales by category
category_sales["total_sales"].plot(kind="bar")

# Add title and axis labels
plt.title("Total Sales by Product Category")
plt.xlabel("Product Category")
plt.ylabel("Total Sales")

# Rotate category labels
plt.xticks(rotation=45)

# Adjust layout and display graph
plt.tight_layout()
plt.show()


# B. Sales by Shopping Mall

# Set graph size
plt.figure(figsize=(12, 5))

# Create bar chart for total sales by shopping mall
mall_sales["total_sales"].plot(kind="bar")

# Add title and axis labels
plt.title("Total Sales by Shopping Mall")
plt.xlabel("Shopping Mall")
plt.ylabel("Total Sales")

# Rotate mall names
plt.xticks(rotation=45)

# Adjust layout and display graph
plt.tight_layout()
plt.show()


# C. Payment Method Sales Distribution

# Set graph size
plt.figure(figsize=(7, 7))

# Create pie chart showing percentage of sales by payment method
payment_sales.plot(
    kind="pie",
    autopct="%1.1f%%",
    startangle=90
)

# Add graph title
plt.title("Sales Distribution by Payment Method")

# Remove y-axis label
plt.ylabel("")

# Adjust layout and display graph
plt.tight_layout()
plt.show()


# D. Monthly Sales Trend

# Set graph size
plt.figure(figsize=(12, 5))

# Create line graph showing monthly sales trend
monthly_sales.plot(kind="line", marker="o")

# Add title and axis labels
plt.title("Monthly Sales Trend")
plt.xlabel("Month")
plt.ylabel("Total Sales")

# Rotate month labels
plt.xticks(rotation=45)

# Display grid lines
plt.grid(True)

# Adjust layout and display graph
plt.tight_layout()
plt.show()


# E. Gender-wise Sales

# Set graph size
plt.figure(figsize=(7, 5))

# Create bar chart showing sales by gender
gender_sales.plot(kind="bar")

# Add title and axis labels
plt.title("Gender-wise Sales")
plt.xlabel("Gender")
plt.ylabel("Total Sales")

# Keep x-axis labels horizontal
plt.xticks(rotation=0)

# Adjust layout and display graph
plt.tight_layout()
plt.show()


# --------------------------------------------------
# 14. EXPORT AGGREGATED DATA
# --------------------------------------------------

# Save category-wise sales into CSV file
category_sales.to_csv("category_sales.csv")

# Save shopping mall-wise sales into CSV file
mall_sales.to_csv("mall_sales.csv")

# Save payment method-wise sales into CSV file
payment_sales.to_csv("payment_sales.csv")

# Save monthly sales into CSV file
monthly_sales.to_csv("monthly_sales.csv")

# Save shopping mall vs category pivot table
pivot_table.to_csv("mall_category_pivot.csv")

# Export final cleaned dataset
data.to_csv("cleaned_customer_shopping_data.csv", index=False)

print("\nAll aggregated files exported successfully!")

print("\nPractical 6 Completed Successfully!")
