# PRACTICAL 4: DATA WRANGLING ON REAL ESTATE MARKET

# 1. Import Libraries
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import seaborn as sns

# Set the style of graphs
sns.set_theme(style="whitegrid")


# 2. Load Dataset
# Read the Bengaluru house dataset
df = pd.read_csv("Bengaluru_House_Data.csv")

print("Dataset Loaded Successfully!")


# 3. Explore Dataset

# Display first 5 rows
print("\nFirst 5 Records:")
print(df.head())

# Display number of rows and columns
print("\nDataset Shape:")
print(df.shape)

# Display column names, data types and non-null values
print("\nDataset Information:")
df.info()

# Count missing values in each column
print("\nMissing Values:")
print(df.isnull().sum())

# Count duplicate records
print("\nDuplicate Records:")
print(df.duplicated().sum())

# Display statistical summary of the dataset
print("\nStatistical Summary:")
print(df.describe(include="all"))


# 4. Remove Duplicate Records
# Remove duplicate rows permanently from the DataFrame
df.drop_duplicates(inplace=True)

print("\nDuplicate Records After Removal:")
print(df.duplicated().sum())


# 5. Remove Unnecessary Columns
# Remove society and availability columns as they are not required
df.drop(
    columns=["society", "availability"],
    inplace=True
)


# 6. Handle Missing Values

# Fill missing location values with the most frequent location (mode)
df["location"] = df["location"].fillna(
    df["location"].mode()[0]
)

# Fill missing size values with the most frequent size
df["size"] = df["size"].fillna(
    df["size"].mode()[0]
)

# Fill missing bathroom values with the median
df["bath"] = df["bath"].fillna(
    df["bath"].median()
)

# Fill missing balcony values with the median
df["balcony"] = df["balcony"].fillna(
    df["balcony"].median()
)


# 7. Extract BHK from Size Column

# Extract numeric values from size column (e.g., 2 from 2 BHK)
df["bhk"] = df["size"].str.extract(r"(\d+)")

# Convert extracted BHK values into numeric format
df["bhk"] = pd.to_numeric(
    df["bhk"], errors="coerce"
)

# Fill any missing BHK values with the median
df["bhk"] = df["bhk"].fillna(
    df["bhk"].median()
)


# 8. Convert Total Square Feet into Numeric Format

# Define a function to convert square feet values into numbers
def convert_sqft(value):
    try:
        # Convert value to string and remove extra spaces
        value = str(value).strip()

        # If value is a range, calculate its average
        if "-" in value:
            low, high = value.split("-")
            return (float(low) + float(high)) / 2

        # If value is a single number, convert it to float
        return float(value)

    # If conversion fails, return NaN
    except:
        return np.nan

# Apply the function to every value in total_sqft column
df["total_sqft"] = df["total_sqft"].apply(
    convert_sqft
)

# Fill invalid or missing square feet values with the median
df["total_sqft"] = df["total_sqft"].fillna(
    df["total_sqft"].median()
)


# 9. Feature Engineering: Price Per Square Foot

# Calculate price per square foot
# Price is multiplied by 100000 because it is given in lakhs
df["price_per_sqft"] = (
    df["price"] * 100000 / df["total_sqft"]
)


# 10. Detect and Handle Outliers Using IQR

# Calculate the first quartile (25th percentile)
Q1 = df["price_per_sqft"].quantile(0.25)

# Calculate the third quartile (75th percentile)
Q3 = df["price_per_sqft"].quantile(0.75)

# Calculate Interquartile Range
IQR = Q3 - Q1

# Calculate lower and upper limits for outliers
lower_bound = Q1 - 1.5 * IQR
upper_bound = Q3 + 1.5 * IQR

# Select records that are outside the limits
outliers = df[
    (df["price_per_sqft"] < lower_bound) |
    (df["price_per_sqft"] > upper_bound)
]

# Display number of outliers before handling
print("\nOutliers Before Handling:", len(outliers))

# Remove records having price per square foot outside the limits
df = df[
    (df["price_per_sqft"] >= lower_bound) &
    (df["price_per_sqft"] <= upper_bound)
].copy()

print("Outlier Handling Completed!")


# 11. Encode Categorical Columns

# Convert area_type categories into numeric codes
df["area_type_encoded"] = df["area_type"].astype(
    "category"
).cat.codes

# Convert location categories into numeric codes
df["location_encoded"] = df["location"].astype(
    "category"
).cat.codes


# 12. Aggregate Data by Location

# Group properties by location and calculate summary statistics
location_summary = df.groupby("location").agg(
    # Calculate average property price
    Average_Price=("price", "mean"),

    # Calculate average area
    Average_Area=("total_sqft", "mean"),

    # Calculate average BHK
    Average_BHK=("bhk", "mean"),

    # Count number of properties in each location
    Property_Count=("location", "count")
).reset_index()

# Display summary of first 10 locations
print("\nLocation-wise Real Estate Summary:")
print(location_summary.head(10))


# 13. Visualize Top 10 Locations by Average Property Price

# Consider only locations having at least 10 properties
filtered_locations = location_summary[
    location_summary["Property_Count"] >= 10
]

# Sort locations by average price in descending order
# Select the top 10 locations
top_locations = filtered_locations.sort_values(
    by="Average_Price",
    ascending=False
).head(10)

# Set the figure size
plt.figure(figsize=(10, 6))

# Create a horizontal bar plot
sns.barplot(
    data=top_locations,
    x="Average_Price",
    y="location"
)

# Add graph title and axis labels
plt.title("Top 10 Locations by Average Property Price")
plt.xlabel("Average Price (Lakhs)")
plt.ylabel("Location")

# Adjust layout and display graph
plt.tight_layout()
plt.show()


# 14. Visualize Property Distribution by Area Type

# Set figure size
plt.figure(figsize=(8, 5))

# Create a count plot to show number of properties in each area type
sns.countplot(
    data=df,
    x="area_type"
)

# Add graph title and axis labels
plt.title("Property Distribution by Area Type")
plt.xlabel("Area Type")
plt.ylabel("Number of Properties")

# Rotate x-axis labels for better readability
plt.xticks(rotation=30)

# Adjust layout and display graph
plt.tight_layout()
plt.show()


# 15. Correlation Heatmap

# Select only numerical columns from the dataset
numeric_df = df.select_dtypes(
    include=np.number
)

# Set figure size
plt.figure(figsize=(9, 6))

# Create a heatmap showing correlation between numerical columns
sns.heatmap(
    numeric_df.corr(),
    annot=True,
    cmap="coolwarm",
    fmt=".2f"
)

# Add title, adjust layout and display graph
plt.title("Correlation Heatmap of Real Estate Features")
plt.tight_layout()
plt.show()


# 16. Remove Duplicate Records After Transformations

# Remove duplicate records again after data transformations
df.drop_duplicates(inplace=True)

# Display remaining duplicate count
print("\nFinal Duplicate Records:")
print(df.duplicated().sum())


# 17. Export Cleaned Dataset

# Save cleaned dataset into a CSV file
df.to_csv(
    "Cleaned_RealEstate_Data.csv",
    index=False
)


# Export Encoded Dataset

# Convert categorical columns into dummy/one-hot encoded columns
encoded_df = pd.get_dummies(
    df,
    columns=["area_type", "location"],
    drop_first=True,
    dtype=int
)

# Save encoded dataset into a CSV file
encoded_df.to_csv(
    "Encoded_RealEstate_Data.csv",
    index=False
)


# Export Location-wise Summary

# Save location-wise summary into a CSV file
location_summary.to_csv(
    "Location_Wise_RealEstate_Summary.csv",
    index=False
)

print("\nAll Files Exported Successfully!")


# 18. Final Verification

# Display final number of rows and columns
print("\nFinal Dataset Shape:")
print(df.shape)

# Display total number of remaining missing values
print("\nRemaining Missing Values:")
print(df.isnull().sum().sum())

# Display total number of remaining duplicate records
print("\nRemaining Duplicate Records:")
print(df.duplicated().sum())

# Display first 5 rows of the final cleaned dataset
print("\nFinal Cleaned Dataset:")
print(df.head())

print("\nData Wrangling Completed Successfully!")
