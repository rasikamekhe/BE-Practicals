# Practical 3: Data Cleaning and Preparation
# Dataset: Telcom_Customer_Churn(1).csv

# Import required libraries
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

# Import function to split dataset into training and testing sets
from sklearn.model_selection import train_test_split

# Import StandardScaler to scale numerical features
from sklearn.preprocessing import StandardScaler


# 1. Load the dataset

# Read the customer churn CSV file into a DataFrame
df = pd.read_csv("Telcom_Customer_Churn(1).csv")

print("Dataset Loaded Successfully!")


# 2. Explore the dataset

# Display the first 5 records
print("\nFirst 5 Records:")
print(df.head())

# Display the number of rows and columns
print("\nDataset Shape:")
print(df.shape)

# Display column names, data types and non-null values
print("\nDataset Information:")
df.info()

# Count missing values in each column
print("\nMissing Values:")
print(df.isnull().sum())

# Count duplicate records
print("\nDuplicate Records:")
print(df.duplicated().sum())

# Display statistical summary of numerical columns
print("\nStatistical Summary:")
print(df.describe())


# 3. Remove duplicate records

# Remove duplicate rows from the dataset
# inplace=True modifies the original DataFrame
df.drop_duplicates(inplace=True)


# 4. Convert TotalCharges into numeric format

# Convert TotalCharges column into numeric data type
# errors="coerce" converts invalid values into NaN
df["TotalCharges"] = pd.to_numeric(
    df["TotalCharges"], errors="coerce"
)

# Check missing values created after conversion
print("\nMissing Values After Conversion:")
print(df.isnull().sum())


# 5. Handle missing values

# Replace missing TotalCharges values with the median
# Median is the middle value of sorted data
df["TotalCharges"] = df["TotalCharges"].fillna(
    df["TotalCharges"].median()
)


# 6. Create a new feature: Tenure Group

# Divide customers into groups based on their tenure
# -1 to 12 months  : New
# 12 to 24 months  : Medium
# 24 to 48 months  : Long
# 48 to 72 months  : Very Long

df["Tenure_Group"] = pd.cut(
    df["tenure"],
    bins=[-1, 12, 24, 48, 72],
    labels=["New", "Medium", "Long", "Very Long"]
)


# 7. Separate numerical and categorical columns

# Select columns containing integer and float values
numerical_cols = df.select_dtypes(
    include=["int64", "float64"]
).columns.tolist()

# Select columns containing text and category values
categorical_cols = df.select_dtypes(
    include=["object", "category"]
).columns.tolist()


# 8. Handle missing values in categorical columns

# Iterate through all categorical columns
for col in categorical_cols:

    # Check whether the column contains missing values
    if df[col].isnull().sum() > 0:

        # Replace missing values with the most frequent value (mode)
        df[col] = df[col].fillna(df[col].mode()[0])

# Check missing values after cleaning
print("\nMissing Values After Cleaning:")
print(df.isnull().sum())


# 9. Check customer churn distribution

# Count customers who left and customers who stayed
print("\nCustomer Churn Distribution:")
print(df["Churn"].value_counts())


# 10. Visualize customer churn distribution

# Create a bar chart showing customer churn
# value_counts() counts Yes and No values
df["Churn"].value_counts().plot(
    kind="bar",
    color=["skyblue", "orange"]
)

# Set graph title
plt.title("Customer Churn Distribution")

# Set X-axis label
plt.xlabel("Churn")

# Set Y-axis label
plt.ylabel("Number of Customers")

# Keep X-axis labels horizontal
plt.xticks(rotation=0)

# Display the graph
plt.show()


# 11. Prepare data for machine learning

# Remove customerID and Churn columns from input features
# customerID is an identifier
# Churn is the target/output column
X = df.drop(["customerID", "Churn"], axis=1)

# Store Churn column separately as target variable
y = df["Churn"]

# Convert categorical variables into numerical values
# drop_first=True avoids redundant dummy columns
X = pd.get_dummies(X, drop_first=True)

# Display the shape of encoded features
print("\nEncoded Feature Shape:")
print(X.shape)


# 12. Split data into training and testing sets

# Split the dataset into 75% training and 25% testing data
X_train, X_test, y_train, y_test = train_test_split(
    X,
    y,
    test_size=0.25,     # 25% data for testing
    random_state=42,    # Ensures reproducible splitting
    stratify=y          # Maintains the same target class proportions
)

# Display training and testing dataset shapes
print("\nTraining Data Shape:", X_train.shape)
print("Testing Data Shape:", X_test.shape)


# 13. Scale numerical features

# Create a StandardScaler object
scaler = StandardScaler()

# Select numerical columns that are present in X_train
numeric_features = [
    col for col in numerical_cols
    if col in X_train.columns
]

# Fit the scaler on training data and transform it
# StandardScaler converts values to mean 0 and standard deviation 1
X_train[numeric_features] = scaler.fit_transform(
    X_train[numeric_features]
)

# Use the same scaler to transform testing data
# Do not fit the scaler again on testing data
X_test[numeric_features] = scaler.transform(
    X_test[numeric_features]
)


# 14. Display final prepared data

# Display the first five rows of prepared training data
print("\nSample Prepared Data:")
print(X_train.head())

# Display completion message
print("\nData Cleaning and Preparation Completed Successfully!")

# Display final training and testing shapes
print("Final Training Shape:", X_train.shape)
print("Final Testing Shape:", X_test.shape)
