# Install required libraries separately in terminal:
# pip install pandas numpy matplotlib openpyxl

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

# ------------------------------------------------
# 1. Load the original CSV dataset
# ------------------------------------------------

# Specify the path of the sales dataset
file_path = "sales_data_sample.csv"

# Read the CSV file using pandas
# cp1252 handles special characters in the dataset
sales = pd.read_csv(file_path, encoding="cp1252")

# Display a success message
print("Dataset Loaded Successfully!")

# Display the first five rows of the dataset
print("\nFirst Five Rows:")
print(sales.head())

# Display the number of rows and columns
print("\nDataset Shape:")
print(sales.shape)

# Display the data types of each column
print("\nData Types:")
print(sales.dtypes)


# ------------------------------------------------
# 2. Check missing values and duplicates
# ------------------------------------------------

# Count missing values in each column
print("\nMissing Values:")
print(sales.isnull().sum())

# Count the number of duplicate records
print("\nDuplicate Records:")
print(sales.duplicated().sum())


# ------------------------------------------------
# 3. Demonstrate CSV, Excel and JSON formats
#    Split the original dataset into three parts.
# ------------------------------------------------

# Split the dataset into three parts using row indexes
part1 = sales.iloc[:941]
part2 = sales.iloc[941:1882]
part3 = sales.iloc[1882:]

# Save the first part in CSV format
part1.to_csv("sales_csv.csv", index=False)

# Save the second part in Excel format
part2.to_excel("sales_excel.xlsx", index=False)

# Save the third part in JSON format
part3.to_json("sales_json.json", orient="records", indent=4)

# Load the saved CSV file
csv_data = pd.read_csv("sales_csv.csv", encoding="cp1252")

# Load the saved Excel file
excel_data = pd.read_excel("sales_excel.xlsx")

# Load the saved JSON file
json_data = pd.read_json("sales_json.json")

# Display the first five records of CSV data
print("\nCSV Data:")
print(csv_data.head())

# Display the first five records of Excel data
print("\nExcel Data:")
print(excel_data.head())

# Display the first five records of JSON data
print("\nJSON Data:")
print(json_data.head())


# ------------------------------------------------
# 4. Combine the datasets
# ------------------------------------------------

# Combine all three datasets into one DataFrame
# ignore_index=True creates a new continuous index
sales = pd.concat(
    [csv_data, excel_data, json_data],
    ignore_index=True
)

# Display complete information about the combined dataset
print("\nCombined Dataset Information:")
sales.info()

# Display the shape of the combined dataset
print("\nCombined Dataset Shape:")
print(sales.shape)


# ------------------------------------------------
# 5. Data Cleaning
# ------------------------------------------------

# Remove duplicate records from the dataset
sales.drop_duplicates(inplace=True)

# Remove unnecessary spaces from column names
sales.columns = sales.columns.str.strip()

# Select all text/object columns and remove extra spaces
for col in sales.select_dtypes(include="object").columns:
    sales[col] = sales[col].str.strip()

# Define the columns that should contain numeric values
numeric_columns = [
    "QUANTITYORDERED",
    "PRICEEACH",
    "SALES",
    "MSRP"
]

# Convert the selected columns into numeric data types
# errors="coerce" converts invalid values into NaN
for col in numeric_columns:
    sales[col] = pd.to_numeric(
        sales[col], errors="coerce"
    )

# Convert ORDERDATE column into datetime format
# Invalid dates are converted into NaT
sales["ORDERDATE"] = pd.to_datetime(
    sales["ORDERDATE"], errors="coerce"
)

# Fill missing numeric values using the median
for col in numeric_columns:
    sales[col] = sales[col].fillna(
        sales[col].median()
    )

# Fill missing categorical/text values with "Unknown"
for col in sales.select_dtypes(include="object").columns:
    sales[col] = sales[col].fillna("Unknown")

# Fill missing dates using the most frequently occurring date
sales["ORDERDATE"] = sales["ORDERDATE"].fillna(
    sales["ORDERDATE"].mode()[0]
)

# Display the first five rows of cleaned data
print("\nCleaned Sales Data:")
print(sales.head())

# Check whether any missing values remain
print("\nMissing Values After Cleaning:")
print(sales.isnull().sum())


# ------------------------------------------------
# 6. Descriptive Statistics
# ------------------------------------------------

# Generate statistical information such as:
# count, mean, standard deviation, minimum, maximum and quartiles
print("\nDescriptive Statistics:")
print(sales.describe())


# ------------------------------------------------
# 7. Sales Analysis
# ------------------------------------------------

# Calculate the total sales of all products
print("\nTotal Sales:", sales["SALES"].sum())

# Calculate the average sales
print("Average Sales:", sales["SALES"].mean())

# Group sales according to product line
# Calculate total sales for each product line
# Sort the results in descending order
product_sales = sales.groupby(
    "PRODUCTLINE"
)["SALES"].sum().sort_values(ascending=False)

# Display sales of each product line
print("\nSales by Product Line:")
print(product_sales)

# Group sales according to country
# Calculate total sales for each country
# Sort the results in descending order
country_sales = sales.groupby(
    "COUNTRY"
)["SALES"].sum().sort_values(ascending=False)

# Display sales of each country
print("\nSales by Country:")
print(country_sales)


# ------------------------------------------------
# 8. Data Visualization
# ------------------------------------------------

# Create two plots side by side
# figsize defines the width and height of the figure
fig, axes = plt.subplots(1, 2, figsize=(14, 5))

# -------------------------------
# Bar Chart
# -------------------------------

# Create a bar chart showing sales by product line
product_sales.plot(
    kind="bar",
    ax=axes[0],
    title="Sales by Product Line"
)

# Set the X-axis label
axes[0].set_xlabel("Product Line")

# Set the Y-axis label
axes[0].set_ylabel("Total Sales")

# Rotate X-axis labels by 45 degrees
axes[0].tick_params(axis="x", rotation=45)

# -------------------------------
# Pie Chart
# -------------------------------

# Create a pie chart showing the percentage distribution
# autopct displays percentage values on each slice
product_sales.plot(
    kind="pie",
    autopct="%1.1f%%",
    ax=axes[1],
    title="Product Line Sales Distribution"
)

# Remove the default Y-axis label
axes[1].set_ylabel("")

# Adjust spacing between the plots
plt.tight_layout()

# Display both charts
plt.show()


# ------------------------------------------------
# 9. Save the cleaned dataset
# ------------------------------------------------

# Save the cleaned dataset in CSV format
sales.to_csv("cleaned_sales_data.csv", index=False)

# Save the cleaned dataset in Excel format
sales.to_excel("cleaned_sales_data.xlsx", index=False)

# Display success message
print("\nCleaned Sales Data Saved Successfully!")

# Display completion message
print("\nSales Data Analysis Completed Successfully!")
