
# PRACTICAL 5: DATA VISUALIZATION USING MATPLOTLIB
# Air Quality Analysis

# 1. Import Libraries
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import seaborn as sns

# 2. Load Dataset
data = pd.read_csv("City_Air_Quality.csv", sep=";")

print("Dataset Loaded Successfully!")

# 3. Remove Empty Columns and Rows
data = data.dropna(axis=1, how="all")
data = data.dropna(how="all")

# Remove extra spaces from column names
data.columns = data.columns.str.strip()

# 4. Explore Dataset
print("\nFirst 5 Records:")
print(data.head())

print("\nDataset Shape:")
print(data.shape)

print("\nDataset Information:")
data.info()

print("\nMissing Values:")
print(data.isnull().sum())

print("\nStatistical Summary:")
print(data.describe())


# 5. Clean and Prepare Data

# Replace decimal commas with decimal points
numeric_cols = data.columns.drop(["Date", "Time"])

for col in numeric_cols:
    data[col] = pd.to_numeric(
        data[col].astype(str).str.replace(",", ".", regex=False),
        errors="coerce"
    )

# Replace -200 with NaN (missing values)
data.replace(-200, np.nan, inplace=True)

# Combine Date and Time columns
data["DateTime"] = pd.to_datetime(
    data["Date"] + " " + data["Time"],
    format="%d/%m/%Y %H.%M.%S",
    errors="coerce"
)

# Remove invalid dates
data.dropna(subset=["DateTime"], inplace=True)

# Sort records by date and time
data.sort_values("DateTime", inplace=True)

# Fill missing numeric values with median
for col in numeric_cols:
    data[col] = data[col].fillna(data[col].median())

print("\nMissing Values After Cleaning:")
print(data.isnull().sum())

print("\nData Types After Cleaning:")
print(data.dtypes)


# 6. Select Important Pollutants
pollutants = [
    "CO(GT)",
    "NMHC(GT)",
    "C6H6(GT)",
    "NOx(GT)",
    "NO2(GT)"
]

# 7. Visualize CO Trend Over Time
plt.figure(figsize=(10, 5))

plt.plot(
    data["DateTime"],
    data["CO(GT)"],
    color="blue",
    label="CO"
)

plt.title("Carbon Monoxide (CO) Trend Over Time")
plt.xlabel("Date")
plt.ylabel("CO Concentration")
plt.legend()
plt.xticks(rotation=45)
plt.grid(True)
plt.tight_layout()
plt.show()

# 8. Visualize Individual Pollutant Trends
for pollutant in pollutants:

    plt.figure(figsize=(10, 4))

    plt.plot(
        data["DateTime"],
        data[pollutant],
        label=pollutant
    )

    plt.title(pollutant + " Trend Over Time")
    plt.xlabel("Date")
    plt.ylabel("Concentration")
    plt.legend()
    plt.xticks(rotation=45)
    plt.grid(True)
    plt.tight_layout()
    plt.show()

# 9. Bar Chart: Average Pollutant Concentrations
average_values = data[pollutants].mean()

plt.figure(figsize=(9, 5))

plt.bar(
    average_values.index,
    average_values.values,
    color="skyblue"
)

plt.title("Average Pollutant Concentrations")
plt.xlabel("Pollutants")
plt.ylabel("Average Concentration")
plt.xticks(rotation=30)
plt.tight_layout()
plt.show()

# 10. Box Plot: Pollutant Distribution
plt.figure(figsize=(10, 5))

plt.boxplot(
    [data[col] for col in pollutants],
    tick_labels=pollutants
)

plt.title("Distribution of Air Pollutants")
plt.xlabel("Pollutants")
plt.ylabel("Concentration")
plt.xticks(rotation=30)
plt.grid(axis="y")
plt.tight_layout()
plt.show()

# 11. Scatter Plot: CO vs NO2
plt.figure(figsize=(7, 5))

plt.scatter(
    data["CO(GT)"],
    data["NO2(GT)"],
    color="green",
    alpha=0.5
)

plt.title("CO vs NO2 Concentration")
plt.xlabel("CO Concentration")
plt.ylabel("NO2 Concentration")
plt.grid(True)
plt.tight_layout()
plt.show()

# 12. Scatter Plot: NOx vs NO2
plt.figure(figsize=(7, 5))

plt.scatter(
    data["NOx(GT)"],
    data["NO2(GT)"],
    color="orange",
    alpha=0.5
)

plt.title("NOx vs NO2 Concentration")
plt.xlabel("NOx Concentration")
plt.ylabel("NO2 Concentration")
plt.grid(True)
plt.tight_layout()
plt.show()

# 13. Correlation Heatmap
plt.figure(figsize=(10, 6))

sns.heatmap(
    data[pollutants].corr(),
    annot=True,
    cmap="coolwarm",
    fmt=".2f"
)

plt.title("Correlation Between Air Pollutants")
plt.tight_layout()
plt.show()

# 14. Monthly Average Pollutant Analysis
data["Month"] = data["DateTime"].dt.month

monthly_data = data.groupby("Month")[pollutants].mean()

print("\nMonthly Average Pollutant Values:")
print(monthly_data)

monthly_data.plot(
    kind="line",
    figsize=(10, 5),
    marker="o"
)

plt.title("Monthly Average Pollutant Trends")
plt.xlabel("Month")
plt.ylabel("Average Concentration")
plt.legend(title="Pollutants")
plt.grid(True)
plt.tight_layout()
plt.show()

# 15. Save Cleaned Dataset
data.to_csv(
    "Cleaned_City_Air_Quality.csv",
    index=False
)

# 16. Final Results
print("\nFinal Dataset Shape:")
print(data.shape)

print("\nRemaining Missing Values:")
print(data.isnull().sum().sum())

print("\nFinal Dataset:")
print(data.head())

print("\nAir Quality Data Visualization Completed Successfully!")
