# Step 1: Import required libraries
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

# Used to standardize the dataset
from sklearn.preprocessing import StandardScaler

# Used to perform Principal Component Analysis
from sklearn.decomposition import PCA


# Step 2: Load the dataset
data = pd.read_csv("Wine.csv")

# Display the first five rows of the dataset
print("First five rows:")
print(data.head())


# Step 3: Separate features and target
# X contains all input features except Customer_Segment
X = data.drop('Customer_Segment', axis=1)

# y contains the target column (wine class)
y = data['Customer_Segment']


# Step 4: Standardize the data
# Create an object of StandardScaler
scaler = StandardScaler()

# Fit the scaler and transform the input features
# This converts features to mean 0 and standard deviation 1
X_scaled = scaler.fit_transform(X)


# Step 5: Apply PCA
# Reduce the 13 original features into 2 principal components
pca = PCA(n_components=2)

# Fit PCA and transform the standardized data
X_pca = pca.fit_transform(X_scaled)


# Step 6: Create a new DataFrame
# Store the reduced data in two columns: PC1 and PC2
pca_df = pd.DataFrame(
    X_pca,
    columns=["PC1", "PC2"]
)


# Step 7: Display the reduced dataset
print("\nReduced Dataset:")
print(pca_df.head())


# Step 8: Display explained variance ratio
# Shows the variance captured by each principal component
print("\nExplained Variance Ratio:")
print(pca.explained_variance_ratio_)


# Step 9: Calculate total variance
# Adds the variance captured by PC1 and PC2
print("\nTotal Variance:")
print(sum(pca.explained_variance_ratio_))


# Step 10: Visualize the PCA results
# Set the size of the graph
plt.figure(figsize=(8, 5))

# Create a scatter plot
# X_pca[:, 0] represents PC1
# X_pca[:, 1] represents PC2
# c=y assigns colors according to wine class
plt.scatter(
    X_pca[:, 0],
    X_pca[:, 1],
    c=y,
    cmap='viridis'
)

# Add labels to X-axis and Y-axis
plt.xlabel("Principal Component 1")
plt.ylabel("Principal Component 2")

# Add graph title
plt.title("PCA - Wine Dataset")

# Display color bar representing wine classes
plt.colorbar(label="Wine Class")

# Display the final graph
plt.show()
