# Step 1: Import required libraries
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
import seaborn as sns

# Import StandardScaler to standardize the dataset
from sklearn.preprocessing import StandardScaler

# Import KMeans clustering algorithm
from sklearn.cluster import KMeans


# Step 2: Load the Iris dataset
# Read the Iris dataset from the CSV file
data = pd.read_csv('iris.csv')


# Step 3: Explore the dataset
# Display the first five rows
print("First five rows:")
print(data.head())

# Display dataset information such as columns, data types and non-null values
print("\nDataset Information:")
data.info()

# Display statistical information such as mean, standard deviation,
# minimum, maximum and quartiles
print("\nStatistical Description:")
print(data.describe(include='all'))

# Check and count missing values in each column
print("\nMissing Values:")
print(data.isnull().sum())


# Step 4: Remove unnecessary columns and separate features
# Id is not a flower measurement; Species is the actual label.
# We do not use Species because K-Means is unsupervised.
# Keep only the four flower measurement features
# errors='ignore' avoids an error if a column is not present
X = data.drop(columns=['Id', 'Species'], errors='ignore')

# Handle missing numerical values, if any, by replacing them with median
X = X.fillna(X.median(numeric_only=True))

# Display the features used for clustering
print("\nFeatures used for clustering:")
print(X.head())


# Step 5: Scale the features
# Create a StandardScaler object
scaler = StandardScaler()

# Standardize all input features
X_scaled = scaler.fit_transform(X)


# Step 6: Determine optimal K using the Elbow Method
# Create an empty list to store WCSS values
wcss = []

# Try different numbers of clusters from 1 to 10
for i in range(1, 11):

    # Create a K-Means model for the current number of clusters
    kmeans = KMeans(
        n_clusters=i,       # Number of clusters
        init='k-means++',   # Select initial centroids intelligently
        random_state=42,    # Ensure reproducible results
        n_init=10            # Run the algorithm 10 times
    )

    # Train the K-Means model
    kmeans.fit(X_scaled)

    # Store the WCSS value for the current number of clusters
    wcss.append(kmeans.inertia_)


# Plot the Elbow graph
# Set the size of the graph
plt.figure(figsize=(8, 5))

# Plot the number of clusters against WCSS values
plt.plot(range(1, 11), wcss, marker='o')

# Add title and axis labels
plt.title('Elbow Method')
plt.xlabel('Number of Clusters (K)')
plt.ylabel('WCSS')

# Display cluster numbers from 1 to 10 on the X-axis
plt.xticks(range(1, 11))

# Display grid lines
plt.grid(True)

# Display the Elbow graph
plt.show()


# Step 7: Apply K-Means with K = 3
# Create the final K-Means model with 3 clusters
kmeans = KMeans(
    n_clusters=3,        # Divide the data into 3 clusters
    init='k-means++',    # Initialize centroids using k-means++
    random_state=42,     # Ensure reproducible results
    n_init=10            # Run the algorithm 10 times
)

# Train the model and predict the cluster number for each data point
y_kmeans = kmeans.fit_predict(X_scaled)


# Step 8: Display cluster assignments
# Display the cluster number assigned to each flower
print("\nCluster Assignments:")
print(y_kmeans)

# Count the number of flowers in each cluster
# sort_index() arranges cluster labels in ascending order
print("\nCluster Counts:")
print(pd.Series(y_kmeans).value_counts().sort_index())


# Step 9: Visualize the clusters using petal length and width
# Set the size of the graph
plt.figure(figsize=(8, 5))

# Plot data points belonging to Cluster 1
# Column index 2 represents Petal Length
# Column index 3 represents Petal Width
plt.scatter(
    X_scaled[y_kmeans == 0, 2],
    X_scaled[y_kmeans == 0, 3],
    s=50,
    label='Cluster 1'
)

# Plot data points belonging to Cluster 2
plt.scatter(
    X_scaled[y_kmeans == 1, 2],
    X_scaled[y_kmeans == 1, 3],
    s=50,
    label='Cluster 2'
)

# Plot data points belonging to Cluster 3
plt.scatter(
    X_scaled[y_kmeans == 2, 2],
    X_scaled[y_kmeans == 2, 3],
    s=50,
    label='Cluster 3'
)

# Plot the cluster centroids
# Centroids represent the center point of each cluster
plt.scatter(
    kmeans.cluster_centers_[:, 2],
    kmeans.cluster_centers_[:, 3],
    s=200,
    marker='X',
    label='Centroids'
)

# Add title and axis labels
plt.title('K-Means Clustering on Iris Dataset')
plt.xlabel('Petal Length (Scaled)')
plt.ylabel('Petal Width (Scaled)')

# Display cluster names in the legend
plt.legend()

# Display grid lines
plt.grid(True)

# Display the final clustering graph
plt.show()
