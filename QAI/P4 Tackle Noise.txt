# Install the required libraries (run separately in terminal)
# %pip install qiskit qiskit-aer matplotlib


# Step 1: Import required libraries
from qiskit import QuantumCircuit, transpile
from qiskit_aer import AerSimulator


# Step 2: Create a quantum circuit with 5 qubits and 5 classical bits
# 3 qubits are used as data qubits (q0, q1, q2)
# 2 qubits are used as ancilla qubits (q3, q4)
# 5 classical bits store the final measurement results

qc = QuantumCircuit(5, 5)


# Step 3: Prepare the logical state |1>

# Apply X gate to qubit 0
# X gate changes the state from |0> to |1>
# This is the original quantum state we want to protect

qc.x(0)


# Step 4: Encode using 3-qubit repetition code

# Apply CNOT gate with qubit 0 as control and qubit 1 as target
# This copies the computational-basis value to qubit 1
qc.cx(0, 1)

# Apply CNOT gate with qubit 0 as control and qubit 2 as target
# Now all three data qubits represent the logical state |111>
# This provides protection against a single bit-flip error

qc.cx(0, 2)


# Step 5: Introduce a bit-flip error on qubit 1

# Apply X gate to qubit 1
# This intentionally introduces an error
# The encoded state changes from |111> to |101>

qc.x(1)

# Barrier separates the encoding and error stages
# It improves circuit readability
qc.barrier()


# Step 6: Calculate error syndrome using ancilla qubits

# First syndrome measurement:
# Compare qubits 0 and 1 using CNOT gates
# If their values are different, ancilla q3 becomes 1

qc.cx(0, 3)
qc.cx(1, 3)

# Second syndrome measurement:
# Compare qubits 1 and 2
# If their values are different, ancilla q4 becomes 1

qc.cx(1, 4)
qc.cx(2, 4)

# Barrier separates error detection and correction
qc.barrier()


# Step 7: Correct the bit-flip error on qubit 1

# Apply Toffoli (CCX) gate
# Qubits 3 and 4 act as control qubits
# Qubit 1 is the target qubit
#
# If both syndrome qubits are 1,
# the X gate is applied to qubit 1
# This corrects the introduced bit-flip error

qc.ccx(3, 4, 1)

# Barrier separates correction and measurement
qc.barrier()


# Step 8: Measure all three data qubits

# Measure data qubit 0 and store result in classical bit 0
qc.measure(0, 0)

# Measure data qubit 1 and store result in classical bit 1
qc.measure(1, 1)

# Measure data qubit 2 and store result in classical bit 2
qc.measure(2, 2)


# Step 9: Measure both syndrome ancilla qubits

# Measure ancilla qubit 3 and store result in classical bit 3
qc.measure(3, 3)

# Measure ancilla qubit 4 and store result in classical bit 4
qc.measure(4, 4)


# Step 10: Display quantum circuit

print("Quantum Error Correction Circuit:")

# Draw and display the complete quantum circuit
print(qc.draw())


# Step 11: Execute circuit using AerSimulator

# Create a quantum simulator
simulator = AerSimulator()

# Convert the circuit into simulator-compatible instructions
compiled = transpile(qc, simulator)

# Execute the circuit 1000 times
# shots=1000 means 1000 measurements are performed
result = simulator.run(compiled, shots=1000).result()

# Retrieve measurement counts
counts = result.get_counts()


# Step 12: Display output

# Print the final measurement results
print("\nError Correction Output:")
print(counts)
