# Step 1: Import required libraries
from qiskit import QuantumCircuit, transpile
from qiskit_aer import AerSimulator
import numpy as np
import matplotlib.pyplot as plt
import math


# Step 2: Define initial and goal puzzle states
# 0 represents the blank tile
# Initial state: Starting arrangement of the puzzle
# Goal state: Arrangement that we want to achieve

initial_state = (1, 2, 3, 0)
goal_state = (1, 0, 3, 2)


# Step 3: Define valid movements of the blank tile
# Each position represents a tile location:
# 0  1
# 2  3
#
# The dictionary stores the valid neighboring positions
# where the blank tile can move.

neighbors = {
    0: [1, 2],
    1: [0, 3],
    2: [0, 3],
    3: [1, 2]
}


# Step 4: Define production rules
# This function moves the blank tile according to the selected choice.

def apply_production_rule(state, choice):

    # Convert tuple into a list because lists are mutable
    state = list(state)

    # Find the current position of the blank tile (0)
    blank = state.index(0)

    # Select an adjacent position based on the choice (0 or 1)
    next_position = neighbors[blank][choice]

    # Swap the blank tile with the selected neighboring tile
    state[blank], state[next_position] = (
        state[next_position],
        state[blank]
    )

    # Convert the updated list back into a tuple and return it
    return tuple(state)


# Step 5: Generate puzzle states from a path
# This function applies a sequence of production choices
# and stores every intermediate puzzle state.

def generate_path(initial, path):

    # Start from the initial puzzle state
    state = initial

    # Store the initial state in the list
    states = [state]

    # Apply each choice one by one
    for choice in path:

        # Apply the production rule to get the next state
        state = apply_production_rule(state, choice)

        # Store the newly generated state
        states.append(state)

    # Return all the generated states
    return states


# Step 6: Define quantum search parameters

# We use 3 qubits for representing production choices
num_qubits = 3

# Total possible paths = 2^3 = 8
total_states = 2 ** num_qubits

# Store indices of paths that reach the goal state
marked_indices = []

# Store the actual valid paths
valid_paths = []


# Step 7: Find all paths reaching the goal state
# Check all possible 3-bit combinations (8 paths).

for index in range(total_states):

    # Convert the index into a 3-bit binary number
    # Reverse the bits to match the qubit order
    # Each bit represents one production choice
    path = tuple(
        int(bit)
        for bit in format(index, '03b')[::-1]
    )

    # Generate puzzle states for the selected path
    states = generate_path(initial_state, path)

    # Check whether the final state matches the goal state
    if states[-1] == goal_state:

        # Store the index of the valid path
        marked_indices.append(index)

        # Store the valid production path
        valid_paths.append(path)


# Display puzzle information
print("TARRATACA'S QUANTUM PRODUCTION SYSTEM")
print("-------------------------------------")

print("Initial State:", initial_state)
print("Goal State:", goal_state)
print("Total Possible Paths:", total_states)

print("\nValid Goal-Reaching Paths:")

# Display all paths that reach the goal
for path in valid_paths:
    print(path)

# Display the indices of valid paths
print("\nMarked Indices:", marked_indices)


# Step 8: Create quantum circuit
# 4 qubits are used:
# First 3 qubits = Search qubits
# Last qubit = Auxiliary (ancilla) qubit
# 3 classical bits store the measurement results

qc = QuantumCircuit(4, 3)


# Step 9: Prepare auxiliary qubit in |-> state

# Apply X gate to the auxiliary qubit (qubit 3)
qc.x(3)

# Apply Hadamard gate to create the |-> state
# This helps in phase kickback during the oracle operation
qc.h(3)


# Step 10: Apply Hadamard gates to search qubits
# Hadamard gates create an equal superposition
# of all 8 possible production paths.

qc.h([0, 1, 2])


# Step 11: Define quantum oracle
# Oracle marks the valid goal-reaching paths
# by changing their phase.

def apply_oracle(circuit, marked_states):

    # Process each marked (valid) index
    for index in marked_states:

        # Convert the marked index into a 3-bit binary string
        # Reverse the bits to match qubit ordering
        bits = format(index, '03b')[::-1]

        # Apply X gates wherever the bit value is 0
        # This converts the marked state into the |111> state
        for qubit in range(num_qubits):

            if bits[qubit] == '0':
                circuit.x(qubit)

        # Apply a multi-controlled X gate
        # All three search qubits control the auxiliary qubit
        # Phase kickback marks the selected state
        circuit.mcx([0, 1, 2], 3)

        # Apply X gates again to restore the original qubit states
        for qubit in range(num_qubits):

            if bits[qubit] == '0':
                circuit.x(qubit)


# Step 12: Define Grover diffusion operator
# The diffuser amplifies the probability of marked states.

def apply_diffuser(circuit):

    # Apply Hadamard gates to all search qubits
    circuit.h([0, 1, 2])

    # Apply X gates to all search qubits
    circuit.x([0, 1, 2])

    # Apply Hadamard gate to the last search qubit
    circuit.h(2)

    # Apply multi-controlled X gate
    # Qubits 0 and 1 control qubit 2
    circuit.mcx([0, 1], 2)

    # Apply Hadamard gate again
    circuit.h(2)

    # Apply X gates again
    circuit.x([0, 1, 2])

    # Apply Hadamard gates again
    circuit.h([0, 1, 2])


# Step 13: Calculate Grover iterations

# Count the number of valid solutions
number_of_solutions = len(marked_indices)

# Calculate the optimal number of Grover iterations
# Formula:
# Iterations = (pi/4) * sqrt(N/M) - 0.5
# N = Total possible states
# M = Number of valid solutions
# round() converts the result into an integer
# max(0, ...) prevents negative iterations

iterations = max(
    0,
    round(
        (math.pi / 4) *
        math.sqrt(total_states / number_of_solutions)
        - 0.5
    )
)

# Display the calculated iterations
print("\nGrover Iterations:", iterations)


# Step 14: Apply Grover search

# Repeat oracle and diffuser operations
# according to the calculated number of iterations

for _ in range(iterations):

    # Oracle marks valid solutions
    apply_oracle(qc, marked_indices)

    # Diffuser amplifies the probability of valid solutions
    apply_diffuser(qc)


# Step 15: Measure search qubits

# Measure the 3 search qubits
# Store their results in classical bits 0, 1 and 2

qc.measure([0, 1, 2], [0, 1, 2])


# Step 16: Display quantum circuit

print("\nQuantum Circuit:")

# Draw and display the complete quantum circuit
print(qc.draw())


# Step 17: Execute using AerSimulator

# Create the quantum simulator
simulator = AerSimulator()

# Convert the circuit into simulator-compatible instructions
# optimization_level=0 disables circuit optimization
compiled = transpile(
    qc,
    simulator,
    optimization_level=0
)

# Execute the circuit 1000 times
# shots=1000 means 1000 measurements are performed
result = simulator.run(
    compiled,
    shots=1000
).result()


# Step 18: Display raw quantum measurement results

# Get the measurement counts
counts = result.get_counts()

# Display the raw binary measurement results
print("\nRaw Quantum Measurement Results:")
print(counts)


# Step 19: Decode results into production paths

# Qiskit displays classical bits in reverse order
# Therefore, reverse each measured bitstring to get the path

decoded_counts = {}

# Process every measured bitstring and its count
for bitstring, count in counts.items():

    # Convert binary string into a tuple of integer choices
    path = tuple(
        int(bit)
        for bit in bitstring[::-1]
    )

    # Store the path and its measurement frequency
    decoded_counts[path] = count


# Display decoded production paths
print("\nDecoded Path Results:")
print("---------------------")

# Sort paths and display their measurement counts
for path, count in sorted(decoded_counts.items()):

    # Check whether the path is a valid goal-reaching path
    status = (
        "Goal Reached"
        if path in valid_paths
        else "Not a Goal"
    )

    # Display path, frequency and its status
    print(
        f"Path: {path} | "
        f"Measurements: {count} | "
        f"Status: {status}"
    )


# Step 20: Select the most frequently measured path

# Find the path having the maximum measurement count
best_path = max(
    decoded_counts,
    key=decoded_counts.get
)

# Display the selected path
print("\nSelected Path:", best_path)


# Step 21: Verify selected path

# Generate all puzzle states using the selected path
solution_states = generate_path(
    initial_state,
    best_path
)

# Display every step of the puzzle solution
print("\nPuzzle Solution Steps:")

for step, state in enumerate(solution_states):

    # Display step number and corresponding puzzle state
    print(f"Step {step}: {state}")


# Check whether the final state matches the goal state
if solution_states[-1] == goal_state:

    print("\nGoal State Reached Successfully!")

else:

    print("\nGoal State Not Reached.")


# Step 22: Plot decoded measurement results

# Convert each production path into a string for graph labels
labels = [
    ''.join(map(str, path))
    for path in sorted(decoded_counts)
]

# Get the measurement frequency of each path
values = [
    decoded_counts[path]
    for path in sorted(decoded_counts)
]

# Assign green color to valid paths
# Assign steelblue color to invalid paths
bar_colors = [
    'green' if path in valid_paths else 'steelblue'
    for path in sorted(decoded_counts)
]

# Create a figure of size 9 x 5
plt.figure(figsize=(9, 5))

# Create a bar graph
# X-axis = Production paths
# Y-axis = Number of measurements
# Colors distinguish valid and invalid paths
plt.bar(
    labels,
    values,
    color=bar_colors
)

# Set the X-axis label
plt.xlabel("Production Path Descriptor")

# Set the Y-axis label
plt.ylabel("Number of Measurements")

# Set the graph title
plt.title("Tarrataca's Quantum Production System - 3 Puzzle")

# Display horizontal grid lines
plt.grid(axis='y', alpha=0.4)

# Automatically adjust graph spacing
plt.tight_layout()

# Display the final bar graph
plt.show()
